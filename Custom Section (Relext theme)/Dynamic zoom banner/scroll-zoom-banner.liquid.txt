<script>
  (function () {
    function initScrollZoom() {
      const section = document.getElementById('SZBSection-{{ section.id }}');
      if (!section) return;

      let ticking = false;

      function updateProgress() {
        const rect = section.getBoundingClientRect();
        const totalScrollable = section.offsetHeight - window.innerHeight;

        if (totalScrollable <= 0) return;

        // Calculate progress from 0 (top of section hits top of window) to 1 (bottom hits bottom)
        let progress = -rect.top / totalScrollable;
        progress = Math.max(0, Math.min(1, progress));

        section.style.setProperty('--szb-progress', progress.toFixed(4));
        ticking = false;
      }

      function onScroll() {
        if (!ticking) {
          requestAnimationFrame(updateProgress);
          ticking = true;
        }
      }

      window.addEventListener('scroll', onScroll, { passive: true });
      window.addEventListener('resize', onScroll, { passive: true });
      updateProgress();
    }

    if (document.readyState === 'loading') {
      document.addEventListener('DOMContentLoaded', initScrollZoom);
    } else {
      initScrollZoom();
    }
  })();
</script>
