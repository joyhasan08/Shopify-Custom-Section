<script>
  (function () {
    function initParallax() {
      const section = document.getElementById('StorySection-{{ section.id }}');
      const leftImg = document.getElementById('ParallaxLeft-{{ section.id }}');
      const rightImg = document.getElementById('ParallaxRight-{{ section.id }}');

      if (!section || !leftImg || !rightImg) return;

      let ticking = false;
      const speed = {{ section.settings.parallax_speed | default: 30 }};

      function updateParallax() {
        const rect = section.getBoundingClientRect();
        const windowHeight = window.innerHeight;

        // Check if section is currently within view
        if (rect.top <= windowHeight && rect.bottom >= 0) {
          // Normalize scroll progress through section (-1 to 1)
          const centerOffset = (rect.top + rect.height / 2) - (windowHeight / 2);
          const normalized = centerOffset / (windowHeight / 2);

          if (window.innerWidth >= 992) {
            const translateYLeft = normalized * speed;
            const translateYRight = -normalized * speed;

            leftImg.style.transform = `translate3d(0, ${translateYLeft}px, 0)`;
            rightImg.style.transform = `translate3d(0, ${translateYRight}px, 0)`;
          } else {
            leftImg.style.transform = 'none';
            rightImg.style.transform = 'none';
          }
        }
        ticking = false;
      }

      function onScroll() {
        if (!ticking) {
          requestAnimationFrame(updateParallax);
          ticking = true;
        }
      }

      window.addEventListener('scroll', onScroll, { passive: true });
      window.addEventListener('resize', onScroll, { passive: true });
      updateParallax();
    }

    if (document.readyState === 'loading') {
      document.addEventListener('DOMContentLoaded', initParallax);
    } else {
      initParallax();
    }
  })();
</script>
