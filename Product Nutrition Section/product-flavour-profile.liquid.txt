{% comment %}
  Dynamic Product Specifications

  Layout:
  - Black section background
  - Left heading and product description
  - Right white card with product name, specifications, and image

  Specifications:
  - Both labels and values are extracted dynamically
  - Displays up to four distinct specifications
  - No fixed specification names
  - No invented values

  Extraction priority:
  1. Two-column specification tables
  2. Definition lists
  3. Label: value text and leading bold labels
  4. Description headings with supporting content
  5. Description bullets or sentences as excerpts

  Optional metafields:
  custom.profile_heading  — Single line text
  custom.profile_intro    — Multi-line text
  custom.profile_name     — Single line text
  custom.profile_image    — File reference, images only

  Previous flavour image/name/intro metafields remain supported
  as secondary fallbacks.

  No external libraries or API keys.
{% endcomment %}

<script>
  if (!customElements.get('dynamic-product-specifications')) {
    customElements.define(
      'dynamic-product-specifications',
      class extends HTMLElement {
        connectedCallback() {
          if (this.initialized) return;
          this.initialized = true;

          const sourceTemplate = this.querySelector(
            '[data-description-source]'
          );
          const rowTemplate = this.querySelector(
            '[data-specification-template]'
          );
          const list = this.querySelector('[data-specification-list]');

          if (!sourceTemplate || !rowTemplate || !list) return;

          this.labelWords = Number(this.dataset.labelWords) || 5;
          this.valueWords = Number(this.dataset.valueWords) || 28;

          const source = sourceTemplate.content.cloneNode(true);

          source.querySelectorAll(
            'script, style, iframe, noscript, template'
          ).forEach((element) => element.remove());

          const specifications = this.extractSpecifications(source);

          list.replaceChildren();

          specifications.forEach((specification) => {
            const fragment = rowTemplate.content.cloneNode(true);
            const label = fragment.querySelector('.dps-label');
            const value = fragment.querySelector('.dps-value');

            label.textContent = this.shorten(
              specification.label,
              this.labelWords
            );

            value.textContent = this.shorten(
              specification.value,
              this.valueWords
            );

            if (label.textContent !== specification.label) {
              label.title = specification.label;
            }

            if (value.textContent !== specification.value) {
              value.title = specification.value;
            }

            list.appendChild(fragment);
          });

          const emptyNote = this.querySelector('[data-empty-note]');

          if (emptyNote) {
            emptyNote.hidden = specifications.length > 0;
          }
        }

        normalize(value) {
          return String(value || '')
            .replace(/\u00a0/g, ' ')
            .replace(/\s+/g, ' ')
            .trim();
        }

        shorten(value, limit) {
          const text = this.normalize(value);
          const words = text.split(/\s+/);

          if (words.length <= limit) return text;

          return words
            .slice(0, limit)
            .join(' ')
            .replace(/[,;:\-]+$/, '') + '…';
        }

        readableText(node) {
          const clone = node.cloneNode(true);

          if (clone.querySelectorAll) {
            clone.querySelectorAll(
              'p, div, li, br, h1, h2, h3, h4, h5, h6'
            ).forEach((element) => {
              element.appendChild(document.createTextNode(' '));
            });
          }

          return this.normalize(clone.textContent);
        }

        splitPair(text) {
          const clean = this.normalize(text);

          // Only treat a short leading phrase as a label.
          const match = clean.match(
            /^([^:：\n]{2,65})\s*[:：]\s*(.+)$/
          );

          if (!match) return null;

          const label = match[1].trim();
          const value = match[2].trim();

          if (
            label.split(/\s+/).length > 9 ||
            !value ||
            /^https?$/i.test(label)
          ) {
            return null;
          }

          return { label, value };
        }

        extractSpecifications(source) {
          const specifications = [];
          const usedLabels = new Set();
          const usedValues = new Set();

          const add = (label, value) => {
            if (specifications.length >= 4) return;

            label = this.normalize(label)
              .replace(/[:：\s]+$/, '');

            value = this.normalize(value);

            if (
              label.length < 2 ||
              !value ||
              label.length > 160
            ) {
              return;
            }

            const labelKey = label.toLocaleLowerCase();
            const valueKey = value.toLocaleLowerCase();

            if (
              usedLabels.has(labelKey) ||
              usedValues.has(valueKey)
            ) {
              return;
            }

            specifications.push({ label, value });
            usedLabels.add(labelKey);
            usedValues.add(valueKey);
          };

          // 1. Two-column specification tables.
          // Wider tables are skipped because their meaning may depend
          // on column headers, serving sizes, or multiple variants.
          source.querySelectorAll('tr').forEach((row) => {
            if (row.closest('thead')) return;

            const cells = [...row.children].filter((cell) => {
              return cell.matches('th, td');
            });

            if (cells.length !== 2) return;
            if (cells.every((cell) => cell.tagName === 'TH')) return;

            add(
              this.readableText(cells[0]),
              this.readableText(cells[1])
            );
          });

          // 2. Definition lists: <dt>Material</dt><dd>Cotton</dd>.
          source.querySelectorAll('dt').forEach((term) => {
            const definition = term.nextElementSibling;

            if (definition?.tagName === 'DD') {
              add(
                this.readableText(term),
                this.readableText(definition)
              );
            }
          });

          // 3. Label: value content and leading bold labels.
          source.querySelectorAll('li, p').forEach((element) => {
            if (
              element.closest('table, dl') ||
              (element.tagName === 'P' && element.closest('li'))
            ) {
              return;
            }

            // Preserve <br> boundaries for specifications written
            // as several lines inside a single paragraph.
            const clone = element.cloneNode(true);

            clone.querySelectorAll('br').forEach((br) => {
              br.replaceWith(document.createTextNode('\n'));
            });

            const lines = clone.textContent
              .split(/\n+/)
              .map((line) => this.normalize(line))
              .filter(Boolean);

            let foundPair = false;

            lines.forEach((line) => {
              const pair = this.splitPair(line);

              if (pair) {
                foundPair = true;
                add(pair.label, pair.value);
              }
            });

            if (foundPair) return;

            const fullText = this.readableText(element);
            const boldElement = element.querySelector('strong, b');
            const boldText = this.normalize(boldElement?.textContent);

            if (
              boldText &&
              fullText.startsWith(boldText) &&
              boldText.split(/\s+/).length <= 9
            ) {
              const remaining = fullText
                .slice(boldText.length)
                .replace(/^[\s:：–—-]+/, '')
                .trim();

              if (remaining) {
                add(boldText, remaining);
              }
            }
          });

          // 4. Headings with their following description content.
          const headings = [
            ...source.querySelectorAll('h2, h3, h4, h5, h6')
          ].filter((heading) => !heading.closest('table, dl'));

          headings.forEach((heading, index) => {
            if (specifications.length >= 4) return;

            try {
              const range = document.createRange();
              range.setStartAfter(heading);

              if (headings[index + 1]) {
                range.setEndBefore(headings[index + 1]);
              } else {
                range.setEnd(source, source.childNodes.length);
              }

              const content = range.cloneContents();

              // Exclude structured data already handled above.
              content.querySelectorAll('table, dl').forEach((element) => {
                element.remove();
              });

              add(
                this.readableText(heading),
                this.readableText(content)
              );
            } catch (error) {
              // Continue to the optional excerpt fallback.
            }
          });

          // 5. Optional excerpts from unstructured description text.
          // Labels are shortened source phrases, not invented categories.
          if (
            specifications.length < 4 &&
            this.dataset.allowExcerpts === 'true'
          ) {
            const excerptSources = [
              ...source.querySelectorAll('li, p')
            ].filter((element) => {
              return (
                !element.closest('table, dl') &&
                !(element.tagName === 'P' && element.closest('li'))
              );
            });

            const excerptTexts = excerptSources.length
              ? excerptSources.map((element) => this.readableText(element))
              : [this.readableText(source)];

            excerptTexts.forEach((text) => {
              if (specifications.length >= 4) return;
              if (!text || this.splitPair(text)) return;

              const sentences = text
                .split(/([.!?])\s+(?=[A-Z0-9\u00C0-\u024F])/)
                .reduce((result, part, index) => {
                  if (index % 2 === 1 && result.length) {
                    result[result.length - 1] += part;
                  } else if (part) {
                    result.push(part);
                  }

                  return result;
                });

              sentences.forEach((sentence) => {
                if (specifications.length >= 4) return;

                const clean = this.normalize(sentence);
                if (clean.length < 15) return;

                const alreadyCovered = specifications.some((item) => {
                  const existing = (
                    item.label + ' ' + item.value
                  ).toLocaleLowerCase();

                  const candidate = clean.toLocaleLowerCase();

                  return (
                    existing.includes(candidate) ||
                    candidate.includes(item.value.toLocaleLowerCase())
                  );
                });

                if (alreadyCovered) return;

                add(this.shorten(clean, this.labelWords), clean);
              });
            });
          }

          return specifications;
        }
      }
    );
  }
</script>
